% COMPATIBILITA' CON I CONTENUTI ESISTENTI
% Nei tuoi file: section -> chapter, subsection -> section,
% subsubsection -> subsection. La conversione vale solo durante l'input.
% Se in futuro convertirai i file a \chapter, userai il normale \input.
\let\TesiSection\section
\let\TesiSubsection\subsection
\newcommand{\InputCapitolo}[2][]{%
  \begingroup
    \let\section\chapter
    \let\subsection\TesiSection
    \let\subsubsection\TesiSubsection
    \ifstrempty{#1}{}{%
      \renewcommand{\chaptermark}[1]{%
        \markboth{\chaptername\ \thechapter.\ #1}{}}}
    \input{#2}
    \clearpage
  \endgroup
}
\newcommand{\InputIntroduzione}[1]{%
  \begingroup
    % Con \frontmatter, memoir genera un capitolo senza numero,
    % ma inserito nell'indice: stesso comportamento di \introduzione.
    \let\section\chapter
    \let\subsection\TesiSection
    \let\subsubsection\TesiSubsection
    \input{#1}
    \clearpage
  \endgroup
}
